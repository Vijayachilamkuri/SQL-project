-- Display customer names along with the vehicles they rented and rental details.

CREATE VIEW customer_rental_details AS
SELECT
    c.customer_id,
    c.customer_name,
    c.city,
    v.vehicle_name,
    v.vehicle_type,
    r.rental_date,
    r.return_date,
    r.amount_paid
FROM customers c
JOIN rentals r
    ON c.customer_id = r.customer_id
JOIN vehicles v
    ON r.vehicle_id = v.vehicle_id;
    SELECT * FROM customer_rental_details;
 -- Display vehicle details and their current availability status.
CREATE VIEW vehicle_availability AS
SELECT
    vehicle_id,
    vehicle_name,
    vehicle_type,
    daily_rate,
    availability
FROM vehicles;
SELECT * FROM vehicle_availability;
-- Display each customer's total rental amount.
CREATE VIEW customer_total_rentals AS
SELECT
    c.customer_id,
    c.customer_name,
    c.city,
    SUM(r.amount_paid) AS total_rental_amount
FROM customers c
JOIN rentals r
    ON c.customer_id = r.customer_id
GROUP BY
    c.customer_id,
    c.customer_name,
    c.city;
    
SELECT * FROM customer_total_rentals;
-- Display the total revenue generated by each vehicle.
CREATE VIEW vehicle_revenue AS
SELECT
    v.vehicle_id,
    v.vehicle_name,
    v.vehicle_type,
    SUM(r.amount_paid) AS total_revenue
FROM vehicles v
JOIN rentals r
    ON v.vehicle_id = r.vehicle_id
GROUP BY
    v.vehicle_id,
    v.vehicle_name,
    v.vehicle_type;
    
SELECT * FROM vehicle_revenue;
 -- Display payment information along with customer and vehicle details.
CREATE VIEW payment_details AS
SELECT
    p.payment_id,
    c.customer_name,
    v.vehicle_name,
    p.payment_date,
    p.payment_method,
    p.payment_amount
FROM payments p
JOIN rentals r
    ON p.rental_id = r.rental_id
JOIN customers c
    ON r.customer_id = c.customer_id
JOIN vehicles v
    ON r.vehicle_id = v.vehicle_id;
    SELECT * FROM payment_details;
