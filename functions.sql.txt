-- Calculate the rental duration in days for each rental
SELECT
    rental_id,
    rental_date,
    return_date,
    DATEDIFF(return_date, rental_date) AS rental_days
FROM rentals;
-- Find the total payment amount collected through each payment method.
SELECT
    payment_method,
    SUM(payment_amount) AS total_payment
FROM payments
GROUP BY payment_method;

-- Display customer names after removing any leading and trailing spaces.

    customer_id,
    TRIM(customer_name) AS customer_name
FROM customers;
